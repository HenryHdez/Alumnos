!Codigo R1
enable
conf t
 hostname R1

 interface FastEthernet0/0
  description LAN VPC1
  ip address 192.168.10.1 255.255.255.0
  no shutdown

 interface FastEthernet0/1
  description Enlace a R2 Fa0/0
  ip address 10.0.12.1 255.255.255.252
  no shutdown

 router ospf 1
  !ID del ospf
  router-id 1.1.1.1
  !Todas las interfaces en modo pasivo por defecto
  passive-interface default
  !La Fa0/1 no es pasiva
  no passive-interface FastEthernet0/1
  %Indica que redes participan del OSPF y pone la wildcard
  %Wilcard es una mascara para identificar bits que deben coincidir
  network 10.0.12.0 0.0.0.3 area 0
  network 192.168.10.0 0.0.0.255 area 0
end
write memory


!Codigo R2
enable
conf t
 hostname R2

 interface FastEthernet0/0
  description Enlace a R1 Fa0/1
  ip address 10.0.12.2 255.255.255.252
  no shutdown

 interface FastEthernet0/1
  description Enlace a R3 Fa0/1
  ip address 10.0.23.1 255.255.255.252
  no shutdown

 router ospf 1
  router-id 2.2.2.2
  passive-interface default
  no passive-interface FastEthernet0/0
  no passive-interface FastEthernet0/1
  network 10.0.12.0 0.0.0.3 area 0
  network 10.0.23.0 0.0.0.3 area 0
end
write memory
